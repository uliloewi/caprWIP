#!/usr/local/bin/foma -f
# Minimal Proto-Germanic toy transducers.

define ProtoVowel [a|e|i|o|u|æ|ɑ|ɔ|ə|ɛ|ɜ|ʉ|ʊ|ø|œ|y];
define ProtoConsonant [b|d|f|g|h|j|k|l|m|n|p|r|s|t|v|w|x|z|ð|ŋ|ɡ|ɣ|ɪ|ʃ|ʋ|ʌ|ʤ|ʧ];
define ProtoSymbol [ProtoVowel|ProtoConsonant];
define ProtoWord ProtoSymbol*;

define English ProtoWord;
clear stack
regex English;
save stack english.bin

define Dutch ProtoWord;
clear stack
regex Dutch;
save stack dutch.bin

define German ProtoWord;
clear stack
regex German;
save stack german.bin
